table worktree_settings: CREATE TABLE worktree_settings ( session_socket_id TEXT NOT NULL, worktree_id TEXT NOT NULL, settled_delay_ms INTEGER NOT NULL CHECK (settled_delay_ms >= 0), PRIMARY KEY (session_socket_id, worktree_id), FOREIGN KEY (session_socket_id, worktree_id) REFERENCES worktrees (session_socket_id, worktree_id) )
table worktrees: CREATE TABLE worktrees ( session_socket_id TEXT NOT NULL, worktree_id TEXT NOT NULL, activated INTEGER NOT NULL CHECK (activated IN (0, 1)), paused INTEGER NOT NULL CHECK (paused IN (0, 1)), deactivated INTEGER NOT NULL CHECK (deactivated IN (0, 1)), created_at_ms INTEGER NOT NULL CHECK (created_at_ms >= 0), PRIMARY KEY (session_socket_id, worktree_id) )

view delivery_seats: CREATE VIEW delivery_seats AS SELECT d.session_socket_id, d.worktree_id, d.delivery_id, COALESCE(c.state, d.state) AS progress FROM mailbox_deliveries d LEFT JOIN space_envoys e ON e.session_socket_id = d.session_socket_id AND e.worktree_id = d.worktree_id AND e.address_id = d.address_id LEFT JOIN bridged_messages b ON b.session_socket_id = d.session_socket_id AND b.worktree_id = e.origin_space_id AND b.origin_space_id = d.worktree_id AND b.origin_message_id = d.message_id LEFT JOIN mailbox_deliveries c ON c.session_socket_id = b.session_socket_id AND c.worktree_id = b.worktree_id AND c.message_id = b.message_id AND c.address_id = e.origin_address_id
